/* ---------------- DAILY QUEST SYSTEM (UPGRADED) ---------------- */ const QUEST_POOL = [ { title:"Touch grass", prize:10, rarity:"Common" }, { title:"Drink 2 bottles of water", prize:15, rarity:"Common" }, { title:"Clean your room", prize:20, rarity:"Uncommon" }, { title:"Do 20 pushups", prize:25, rarity:"Uncommon" }, { title:"Draw something", prize:30, rarity:"Rare" }, { title:"Talk to someone new", prize:40, rarity:"Rare" }, { title:"Go outside for 10 minutes", prize:50, rarity:"Epic" }, { title:"Do something kind", prize:60, rarity:"Legendary" } ]; function generateDailyQuests() { let quests = []; let used = new Set(); while (quests.length < 3) { let r = Math.floor(Math.random() * QUEST_POOL.length); if (!used.has(r)) { used.add(r); quests.push(QUEST_POOL[r]); } } return quests; } function loadDailyQuests() { const today = new Date().toDateString(); const savedDate = localStorage.getItem("dailyDate"); let dailyQuests; if (savedDate !== today) { dailyQuests = generateDailyQuests(); localStorage.setItem("dailyQuests", JSON.stringify(dailyQuests)); localStorage.setItem("dailyDate", today); } else { dailyQuests = JSON.parse(localStorage.getItem("dailyQuests")); } displayDailyQuests(dailyQuests); } function displayDailyQuests(quests) { const list = document.getElementById("dailyList"); list.innerHTML = ""; quests.forEach((q, index) => { const div = document.createElement("div"); div.className = "quest-box"; div.innerHTML = `

${q.title}

${q.rarity}

Reward: ${q.prize} XP

`; list.appendChild(div); }); } function completeDaily(index) { document.getElementById("collect" + index).style.display = "block"; } function rejectDaily(index) { let quests = JSON.parse(localStorage.getItem("dailyQuests")); quests.splice(index, 1); localStorage.setItem("dailyQuests", JSON.stringify(quests)); loadDailyQuests(); } function collectReward(index) { let quests = JSON.parse(localStorage.getItem("dailyQuests")); let xp = parseInt(localStorage.getItem("xp") || "0"); xp += quests[index].prize; localStorage.setItem("xp", xp); // Move quest to progress let progress = JSON.parse(localStorage.getItem("progress") || "[]"); progress.push(quests[index]); localStorage.setItem("progress", JSON.stringify(progress)); // Remove quest from daily list quests.splice(index, 1); localStorage.setItem("dailyQuests", JSON.stringify(quests)); alert("Reward collected! +" + quests[index]?.prize + " XP"); loadDailyQuests(); }
